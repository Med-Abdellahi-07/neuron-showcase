-- Rollback of 0027 (manual, never run automatically). Restores the spaces' state and the moderators' space from the backup.
update public.cohorts c set is_active = b.is_active, plan = b.plan, features = b.features
  from public.single_space_backup b where b.kind = 'cohort' and b.id = c.id;
update public.profiles p set cohort_id = b.cohort_id
  from public.single_space_backup b where b.kind = 'admin_profile' and b.id = p.id;
